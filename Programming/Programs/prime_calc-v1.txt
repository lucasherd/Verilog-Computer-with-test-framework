LOADI #1, [r0] //current candidate
LOADI #0, [r7] //current memory pointer

CHECK_PRIME:
INC r0, [r0]
CMPI r0, #0
BR LT, END //stop program once reg overflows
SHR r0, [r1] //start checking divisible by n/2

NEXT_FACTOR:
MOV r0, [r2]
LOADI #0, [r3] //number of subractions -> only need to check until n_sub = dividend

MOD:
SUB r2, r1, [r2]
INC r3, [r3]
CMP r3, r1
BR GT, FOUND_PRIME //no factors found and square root passed, therefore prime
CMPI r2, #0
BR GT, MOD
BR EQ, CHECK_PRIME // found divisor, not prime
SUBI r1, #1, [r1]
JMP NEXT_FACTOR

FOUND_PRIME: //r0 is prime, store to mem
STORE r0, m[r7]
INC r7, [r7]
JMP CHECK_PRIME

END:
HALT
